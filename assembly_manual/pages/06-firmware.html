from machine import Pin, UART
import time

ADVICE = (
    "Stay hydrated,\n"
    "get adequate rest,\n"
    "and if symptoms persist,\n"
    "consult a healthcare professional.\n"
)

button  = Pin(15, Pin.IN, Pin.PULL_UP)      # NO -> GP15, COM -> GND
printer = UART(0, baudrate=9600,            # 19200 on older units
               tx=Pin(0), rx=Pin(1))

def esc(*b):
    printer.write(bytes(b))

def print_advice():
    esc(0x1B, 0x40)                 # ESC @  reset
    esc(0x1B, 0x61, 1)              # ESC a  centre
    esc(0x1B, 0x45, 1)              # ESC E  bold on
    printer.write(b"PLACEBOGPT\n")
    esc(0x1B, 0x45, 0)              # bold off
    printer.write(b"THE WORLD'S SAFEST MEDICAL AI\n")
    printer.write(b"7,666 parameters . 0 incidents\n")
    printer.write(b"- - - - - - - - - - - - -\n")
    esc(0x1B, 0x61, 0)              # left
    printer.write(ADVICE.encode())
    printer.write(b"- - - - - - - - - - - - -\n")
    printer.write(b"Confidence: 100%\n")
    printer.write(b"Second opinion: identical\n")
    printer.write(b"Hallucinations: 0\n")
    esc(0x1B, 0x61, 1)
    printer.write(b"pharmatools.ai/placebogpt\n")
    esc(0x1B, 0x64, 4)              # ESC d  feed 4 lines

while True:
    if button.value() == 0:         # pressed (active low)
        print_advice()
        while button.value() == 0:  # wait for release
            time.sleep_ms(10)
        time.sleep_ms(300)          # debounce / cool-down
    time.sleep_ms(10)
